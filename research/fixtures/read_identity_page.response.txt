    function ft (ms)

        }
        if (r > 0) {
            m=r%60; r=parseInt(r/60);

        }
        if (r > 0) {
            h=r%24; r=parseInt(r/24);
            msg = h + (h > 1 ? " hours" : " hour") + ", " + msg;
        }
        if (r > 0) {
            d=r;
            msg = d + (d > 1 ? " days" : " day") + ", " + msg;
        }
        return msg;
    } // ft
